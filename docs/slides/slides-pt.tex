\section{Visão geral}
\begin{frame}{Porquê um logbook?}
  \small
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Hoje}
    \begin{itemize}
      \item Cadernos em papel e\\ folhas de cálculo pessoais
      \item Ficheiros de dados espalhados por\\ PCs de instrumentos e pens USB
      \item ``Que substrato era aquele?\\ A que temperatura foi o RTA?''
      \item Difícil passar amostras a outra pessoa\\ ou encontrar resultados de colegas
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{Com o logbook}
    \begin{itemize}
      \item Um ID único e a história completa\\ de cada amostra
      \item Parâmetros em campos estruturados,\\ por isso pesquisáveis
      \item Fotografias e dados em bruto\\ anexados onde pertencem
      \item Partilhado, pesquisável,\\ exportável para CSV
    \end{itemize}
  \end{columns}
\end{frame}

\section{Registos}
\begin{frame}{Registar um passo ou análise}
  \begin{columns}[T]
    \column{0.5\textwidth}
    \begin{itemize}
      \item Escolha o tipo e o formulário\\ mostra os seus campos
      \item \textbf{Onde} + \textbf{Equipamento}:\\ só as máquinas desse local
      \item Unidades incluídas; \texttt{1e-4} e \texttt{12:30} funcionam
      \item As listas lembram valores;\\ \ui{+ Adicionar novo\ldots} para um novo
      \item \textbf{Realizado por}: quem o fez pode ver\\ e editar essa entrada
    \end{itemize}
    \column{0.46\textwidth}
    \small
    \textbf{Passos:} Deposition, Annealing (Furnace / RTA), Ion
    Implantation, Lithography, Etching, CMP, Other

    \vspace{0.3cm}
    \textbf{Análises:} XRD / HRXRD (with modes), XPS, SEM, TEM, RBS, AFM,
    Raman, Optical Microscopy, Hall / 4-probe, Ellipsometry, Profilometry, Other

    \vspace{0.3cm}
    \footnotesize Os administradores podem acrescentar tipos e campos sem mexer no código.
  \end{columns}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Fotografias e ficheiros de dados}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{\ui{Adicionar fotografias}}
    \begin{itemize}
      \item Diretamente da câmara do telemóvel, ou do disco
      \item Várias de uma vez
      \item Reduzidas a $\leq$\,2\,MB, HEIC convertido, GPS removido
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{\ui{Anexar ficheiros}}
    \begin{itemize}
      \item Dados em bruto, \alert{guardados exatamente como enviados}
      \item 16-bit TIFFs, \texttt{.raw}, \texttt{.csv}, PDFs, \ldots
      \item Miniaturas na cronologia, e uma pasta por amostra no disco, com um \texttt{sample.json} (cópia da sua cronologia)
    \end{itemize}
  \end{columns}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Um ensaio, várias amostras}
  A mesma deposição ou sessão de XRD para várias amostras?
  \begin{itemize}
    \item Preencha o formulário uma vez e marque \ui{Adicionar também a outras amostras}
    \item Cada amostra recebe a sua cópia ligada (``Também em: \ldots'')
    \item As fotografias comuns são guardadas uma vez,\\ e cada amostra tem os seus dados
    \item As amostras irmãs (cortadas do mesmo passo) são sugeridas primeiro
    \item \ui{O que difere entre amostras?}: uma caixa por amostra para o
          parâmetro que muda numa série
    \item As alterações podem aplicar-se a todo o ensaio, com aviso antes
          de substituir valores que outra amostra já tem
  \end{itemize}
\end{frame}

  \vspace{0.3cm}
  \raggedright\small
  \begin{itemize}
    \item Os pedaços \alert{herdam} a história até ao corte (a cinzento):
          uma correção na amostra-mãe chega a todos os pedaços
    \item Depois, cada pedaço tem as suas entradas; pode ir para outra pessoa
    \item Pedaço de referência opcional \sid{-REF}
    \item Em \ui{Amostras}, os pedaços ficam sob a sua amostra
          (\ui{$\triangleright$ N pedaços});\\ \ui{Expandir tudo} abre todos os grupos
  \end{itemize}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Quem vê o quê}
  \begin{columns}[T]
    \column{0.48\textwidth}
    \textbf{Todos}: o resumo de cada amostra
    \begin{itemize}\small
      \item ID, rótulo, tipo, materiais, substrato
      \item etiquetas, técnicas usadas, investigador
    \end{itemize}
    \column{0.48\textwidth}
    \textbf{Detalhes} (notas, ficheiros) só para:
    \begin{itemize}\small
      \item o responsável e os administradores
      \item quem tem partilha: \ui{Ver} ou \ui{Editar}
      \item numa entrada, quem a realizou
    \end{itemize}
  \end{columns}

  \vspace{0.5cm}
  \small As amostras podem passar para outro investigador, que confirma
  com \ui{Aceitar}/\ui{Recusar};\\ o ID mantém-se.
\end{frame}

\section{Utilização}
\begin{frame}{Pesquisa e exportação}
  \begin{itemize}
    \item Filtrar por texto, investigador, tipo, material, técnica, etiqueta
    \item Agrupar por investigador, ano, tipo, material, técnica, etiqueta, projeto
    \item \ui{Exportar resultados (CSV)}: uma linha por passo/análise, uma
          coluna por parâmetro, pronto para Origin, Excel ou pandas
    \item Selecionar várias amostras e partilhá-las de uma vez
  \end{itemize}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Também}
  \begin{itemize}
    \item \textbf{Biblioteca de substratos}: características e fichas técnicas introduzidas uma vez
    \item \textbf{Projetos}: instituições parceiras e as suas amostras
    \item \textbf{Sugestões}: comunicar um erro ou pedir um campo; os administradores respondem
    \item \textbf{Administradores}: acrescentam tipos e utilizadores, podem enviar e-mails aos utilizadores
    \item Português / inglês, modo escuro, funciona no telemóvel
  \end{itemize}
\end{frame}

%---------------------------------------------------------------------
\begin{frame}{Como começar}
  \begin{enumerate}
    \item Peça uma conta a um administrador
    \item Inicie sessão, defina palavra-passe, nome e \alert{iniciais}
    \item Crie uma amostra: \ui{Nova amostra}
    \item Acrescente passos e análises à medida que avança, com fotografias do telemóvel
  \end{enumerate}

  \vspace{0.5cm}
  Os detalhes estão no manual de utilização (\texttt{docs/manual}).

\end{frame}
